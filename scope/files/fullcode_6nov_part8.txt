// creator must claim it (CollectCreatorFeeEvent on pump.fun, CollectCoinCreatorFeeEvent on PumpSwap). Accrued fees come free from
// the trade events; claims are decoded from the same log streams. Everything is recorded, nothing gated — the analysis decides
// whether "dev has claimed" separates runners from rugs.

const accrued = new Map();   // mint -> {sol, n, dirty}

let samples = 0;

function acc(mint, sol) {
  if (!(sol > 0)) return;
  let a = accrued.get(mint); if (!a) { a = { sol: 0, n: 0, dirty: false }; accrued.set(mint, a); if (accrued.size > 30000) accrued.delete(accrued.keys().next().value); }
  a.sol += sol; a.n++; a.dirty = true;
}
on("trade", (t, tr) => { if (tr.creatorFee > 0) acc(t.mint, tr.creatorFee); });
onSwap((tr) => { if (tr.creatorFee > 0) acc(tr.mint, tr.creatorFee); });

on("claim", (_, c) => {
  fe.claims++; if (c.venue === "pump") fe.claimsPump++; else fe.claimsSwap++;
  // v0.18.8: the vault is per creator, so credit the claim to their most recent coin; flag it if they launched more than one in the last 24h
  let mint = null, amb = 0;
  try { const r = q.latestMintOf.get(c.ts - 86400, c.creator, c.ts); if (r) { mint = r.mint; amb = r.n > 1 ? 1 : 0; fe.attributed++; } } catch {}
  try { q.insClaim.run(c.creator, c.ts, c.sol, c.venue, c.sig, mint, amb); } catch {}
  if (samples < 5) { samples++; console.log("[fee claim]", JSON.stringify({ venue: c.venue, creator: c.creator.slice(0, 8), sol: +c.sol.toFixed(4), mint: mint?.slice(0, 8) || null, ambiguous: amb })); }
  if (mint) { let a = accrued.get(mint); if (!a) { a = { sol: 0, n: 0, dirty: true }; accrued.set(mint, a); } a.dirty = true; }
});

// keep a GMGN object small: drop images / long text / big arrays, then cap the size by removing the largest remaining keys
function slim(o, cap = 9000) {
  if (!o || typeof o !== "object") return o;
  const c = JSON.parse(JSON.stringify(o));
  const drop = (x) => { if (!x || typeof x !== "object") return; for (const k of Object.keys(x)) { if (/logo|avatar|banner|description|pfp|image/i.test(k)) delete x[k]; else if (Array.isArray(x[k]) && x[k].length > 20) x[k] = x[k].slice(0, 20); else drop(x[k]); } };
  drop(c);
  let str = JSON.stringify(c);
  while (str.length > cap) { const big = Object.keys(c).filter((k) => !["dev", "stat", "wallet_tags_stat", "price"].includes(k)).sort((a, b) => JSON.stringify(c[b]).length - JSON.stringify(c[a]).length)[0]; if (!big) break; delete c[big]; str = JSON.stringify(c); }
  return c;
}

/** Levels exit tracker: sells LEVEL_SLICE at 97% of each level crossed; remainder trails 60%. state = {sold:[{mult}], } */
export function levelsStep(lv, entryMc, mc, mult, fromPeak) {

  }
  for (const L of CFG.LEVELS_USD) {
    if (L <= entryMc) continue;
    const trigger = L * CFG.LEVEL_SELL_PCT;
    if (mc >= trigger && !lv.crossed.has(L) && lv.sold < 1 - 1e-9) {
      const slice = Math.min(CFG.LEVEL_SLICE, 1 - lv.sold);
      lv.crossed.add(L); lv.sold += slice; lv.realized += slice * (trigger / entryMc);
    }
  }
  if (lv.done == null && fromPeak <= -0.60) lv.done = lv.realized + (1 - lv.sold) * mult;
  return lv.done;
}
