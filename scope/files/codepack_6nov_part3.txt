import { Connection, VersionedTransaction, TransactionMessage, PublicKey } from "@solana/web3.js";
import { CFG } from "./config.mjs";

export const conn = new Connection(CFG.RPC_URL, { commitment: "confirmed", disableRetryOnRateLimit: true });
export const rpcStats = { calls: 0, errors: 0, sends: 0, senderErrors: 0, lastErr: "" };

let bh = { blockhash: null, lastValidBlockHeight: 0, t: 0 }, bhBusy = false;
export async function blockhash(maxAgeMs = 1500) {
  if (bh.blockhash && Date.now() - bh.t < maxAgeMs) return bh;
  if (bhBusy) { await new Promise((r) => setTimeout(r, 50)); return bh.blockhash ? bh : blockhash(maxAgeMs); }
  bhBusy = true; try { rpcStats.calls++; const r = await conn.getLatestBlockhash("confirmed"); bh = { ...r, t: Date.now() }; } catch (e) { rpcStats.errors++; rpcStats.lastErr = e.message; } finally { bhBusy = false; }
  if (!bh.blockhash) throw new Error("no blockhash: " + rpcStats.lastErr); return bh;
}
export async function getAccounts(pubkeys, commitment = "processed") {
  rpcStats.calls++; try { return await conn.getMultipleAccountsInfo(pubkeys.map((p) => (p instanceof PublicKey ? p : new PublicKey(p))), { commitment }); } catch (e) { rpcStats.errors++; rpcStats.lastErr = e.message; throw e; }
}
/** v0.4.12 local stops: push notifications for an account (the bonding curve, or a pool's reserve accounts) at 'processed' commitment - the executor sees the price move within a slot. */

export async function offAccount(id) { try { await conn.removeAccountChangeListener(id); rpcStats.subs = Math.max(0, (rpcStats.subs || 1) - 1); } catch {} }

export async function buildTx(payer, ixs, signer = null, tables = []) {
  const { blockhash: b, lastValidBlockHeight } = await blockhash();
  const msg = new TransactionMessage({ payerKey: payer, recentBlockhash: b, instructions: ixs }).compileToV0Message(tables); // launcher 0.1.15: lookup tables shrink the launch transaction
  const tx = new VersionedTransaction(msg); tx.lastValidBlockHeight = lastValidBlockHeight; if (signer) tx.sign(Array.isArray(signer) ? signer : [signer]); return tx;
}

/** Simulate; `watch` = accounts whose post-simulation state we want back (e.g. our token account, to read the real fill the chain would give). */
export async function simulate(tx, watch = []) {
  const t0 = Date.now(); rpcStats.calls++;
  try {
    const cfg = { sigVerify: false, replaceRecentBlockhash: true, commitment: "processed" }; if (watch.length) cfg.accounts = { encoding: "base64", addresses: watch.map((k) => k.toBase58()) };
    const r = await conn.simulateTransaction(tx, cfg);
    const post = (r.value.accounts || []).map((a) => (a && a.data ? Buffer.from(a.data[0], "base64") : null));
    return { ok: !r.value.err, err: r.value.err ? JSON.stringify(r.value.err) : null, logs: r.value.logs || [], units: r.value.unitsConsumed ?? null, ms: Date.now() - t0, post };
  } catch (e) { rpcStats.errors++; rpcStats.lastErr = e.message; return { ok: false, err: "rpc: " + e.message, logs: [], units: null, ms: Date.now() - t0, post: [] }; }
}
/** Send a signed transaction through Helius Sender (every landing route at once); falls back to the normal RPC. Returns { sig, via, ms }. */
export async function send(tx) {
  const raw = Buffer.from(tx.serialize()).toString("base64"), t0 = Date.now(); rpcStats.sends++;
  try {
    const r = await fetch(CFG.SENDER_URL, { method: "POST", headers: { "content-type": "application/json" }, signal: AbortSignal.timeout(4000),
      body: JSON.stringify({ id: String(Date.now()), jsonrpc: "2.0", method: "sendTransaction", params: [raw, { encoding: "base64", skipPreflight: true, maxRetries: 0 }] }) });
    const j = await r.json(); if (j.result) return { sig: j.result, via: "sender", ms: Date.now() - t0 };
    throw new Error(j.error?.message || `sender http ${r.status}`);
  } catch (e) {
    rpcStats.senderErrors++; rpcStats.lastErr = "sender: " + e.message;
    const sig = await conn.sendRawTransaction(tx.serialize(), { skipPreflight: true, maxRetries: 0 }); return { sig, via: "rpc", ms: Date.now() - t0, senderErr: e.message };
  }
}
/** Send, then keep re-sending the SAME signed transaction every 2.5 s until it is confirmed / failed or `ms` passes. Re-sending the identical
 *  transaction can never execute twice, so this is the only retry a BUY gets (a second, different buy could land as well → bought twice). */
export async function sendUntilConfirmed(tx, ms = 18000) {
  const first = await send(tx); const sig = first.sig, t0 = Date.now(); let resends = 0;
  while (Date.now() - t0 < ms) {
    const c = await waitConfirm(sig, 2500); if (c.status !== "timeout") return { ...c, sig, via: first.via, resends, ms: Date.now() - t0 };
    try { await send(tx); resends++; } catch {}
  }
  return { status: "timeout", slot: null, err: null, sig, via: first.via, resends, ms: Date.now() - t0 };
}
/** Poll until the signature is confirmed or failed, or `ms` passes. Returns { status: 'confirmed'|'failed'|'timeout', slot, err }. */
export async function waitConfirm(sig, ms = 20000) {
  const t0 = Date.now();
  while (Date.now() - t0 < ms) {
    rpcStats.calls++;
    try { const r = await conn.getSignatureStatuses([sig]); const s = r.value[0];
      if (s) { if (s.err) return { status: "failed", slot: s.slot, err: JSON.stringify(s.err) }; if (s.confirmationStatus === "confirmed" || s.confirmationStatus === "finalized") return { status: "confirmed", slot: s.slot, err: null }; } }
    catch (e) { rpcStats.errors++; rpcStats.lastErr = e.message; }
    await new Promise((r) => setTimeout(r, 200)); // v0.4.11: notice the landing sooner (was 400 ms)
  }
  return { status: "timeout", slot: null, err: null };
}

"failed" : (s.confirmationStatus === "confirmed" || s.confirmationStatus === "finalized") ? "confirmed" : "processed", slot: s.slot, err: s.err ? JSON.stringify(s.err) : null } : { status: "unknown" }; }
/** What actually happened: SOL change for the payer (lamports, fee included) and token change for the mint. Retries a few times — the node can lag the confirmation by a moment. */
export async function readFill(sig, payer, mint, tries = 4) {
  let tx = null; for (let i = 0; i < tries && !tx?.meta; i++) { if (i) await new Promise((r) => setTimeout(r, 600)); rpcStats.calls++; tx = await conn.getTransaction(sig, { maxSupportedTransactionVersion: 1, commitment: "confirmed" }).catch(() => null); }
  if (!tx?.meta) return null;
  const keys = tx.transaction.message.staticAccountKeys.map((k) => k.toBase58()), i = keys.indexOf(payer.toBase58());
  const solDelta = i >= 0 ? tx.meta.postBalances[i] - tx.meta.preBalances[i] : null;
  const tok = (arr) => (arr || []).filter((b) => b.mint === mint.toBase58() && b.owner === payer.toBase58()).reduce((a, b) => a + Number(b.uiTokenAmount.amount), 0);
  return { solDelta, tokenDelta: tok(tx.meta.postTokenBalances) - tok(tx.meta.preTokenBalances), fee: tx.meta.fee, slot: tx.slot, blockTime: tx.blockTime ?? null, err: tx.meta.err ? JSON.stringify(tx.meta.err) : null };
}

let watching = false;
async function watch() {
  if (watching) return; watching = true;
  try {
    const open = db.prepare("SELECT * FROM launches WHERE mode='live' AND status IN ('open','selling') ORDER BY id").all(); if (!open.length) return;
    const g = await global(); const accs = await getAccounts(open.map((L) => curveOf(new PublicKey(L.mint))), "confirmed");

      const x = Number(quoteSell(c, g, BigInt(L.tokens || "0"))) / spent; // what our tokens would sell for now ÷ the stake (≈ 0.975 at the start: the 1.25% fees each way)

      const o = { last_x: +x.toFixed(4), peak_x: Math.max(L.peak_x || 0, +x.toFixed(4)) };
      if (others && !L.others) { o.others = 1; o.first_other_s = age; }
      if (changed) o.last_change_ts = now();
      upd(L.id, o);
      if (c.complete) { upd(L.id, { status: "graduated", reason: "graduated — sell on PumpSwap by hand for now" }); continue; }
      const manual = L.status === "selling";
      // 0.1.11 (owner, 31 Oct 01:15 UK): at DEAD_MIN, count the buys by other wallets; fewer than MIN_BUYS → sell back (checked once)
      let buys = L.buys_10m; if (!manual && buys == null && age >= CFG.DEAD_MIN * 60) { buys = await countBuys(L).catch(() => null); if (buys != null) upd(L.id, { buys_10m: buys }); }
      const why = manual ? (L.sell_reason || "sell button") : buys != null && L.buys_10m == null && buys < CFG.MIN_BUYS ? `only ${buys} buy${buys === 1 ? "" : "s"} by others in ${CFG.DEAD_MIN} min (needs ${CFG.MIN_BUYS})`
        : buys == null && !others && !L.others && age >= CFG.DEAD_MIN * 60 ? `nobody else bought in ${CFG.DEAD_MIN} min`
        : (others || L.others) && now() - (changed ? now() : L.last_change_ts || now()) >= CFG.QUIET_MIN * 60 ? `no trades for ${CFG.QUIET_MIN} min`
        : null;
      if (why) await sellOne(L, c, g, why);
      else if (CFG.TP_X > 0 && !L.tp_done && x >= CFG.TP_X) await sellOne(L, c, g, `take-profit: ${Math.round(CFG.TP_FRACTION * 100)}% sold at ${CFG.TP_X}x`, CFG.TP_FRACTION);

/** Buys by other wallets on our coin's curve so far: its transactions, those that ran a pump.fun buy and were not paid for by our wallet. */
async function countBuys(L) {
  const curve = curveOf(new PublicKey(L.mint)); rpcStats.calls++;
  const sigs = (await conn.getSignaturesForAddress(curve, { limit: 200 }, "confirmed")).filter((s) => !s.err && s.signature !== L.sig).map((s) => s.signature);
  let n = 0;
  for (let i = 0; i < sigs.length; i += 25) {
    rpcStats.calls++; const txs = await conn.getTransactions(sigs.slice(i, i + 25), { maxSupportedTransactionVersion: 1, commitment: "confirmed" });
    for (const t of txs) { if (!t?.meta) continue; const payer = t.transaction.message.staticAccountKeys[0].toBase58();

}
async function sellOne(L, curve, g, why, frac = 1) {
  const mint = new PublicKey(L.mint), ua = ata(ME, mint, TOKEN_2022);
  try {
    const [acc] = await getAccounts([ua], "confirmed"); const held = acc?.data?.length >= 72 ? acc.data.readBigUInt64LE(64) : 0n;
    const part = frac < 1, amount = part ? (held * BigInt(Math.round(frac * 1e4))) / 10000n : held;
    if (amount <= 0n) { upd(L.id, { status: "sold", sell_reason: why + " (no tokens left)", sold_ts: now() }); return; }
    const out = quoteSell(curve, g, amount), minSol = (out * BigInt(Math.round((1 - CFG.SELL_SLIP) * 1e4))) / 10000n;
    const sellIxs = L.buy_ix === "v3" ? ixSellV3({ user: ME, mint, g, amount, minSol, close: !part }) : ixSell({ user: ME, mint, curve, g, amount, minSol, close: !part });
    const tx = await buildTx(ME, [...ixBudget(200000, CFG.CU_PRICE), ...sellIxs, ixTransfer(ME, tipAccount(), lam(CFG.TIP_SOL))], [wallet], luts());
    const tries = (L.sell_tries || 0) + 1; if (!part) upd(L.id, { status: "selling", sell_reason: why, sell_tries: tries });
    const sim = await simulate(tx); // a sale the chain would refuse is not sent (a failed transaction still costs its fee)
    const r = sim.ok ? await sendUntilConfirmed(tx, 20000) : { status: "refused", err: sim.err };
    if (part) { // take-profit on part of the position: record it once, keep holding the rest under the normal rules
      if (r.status !== "confirmed") { st.lastErr = `take-profit ${L.ticker}: ${r.status} ${r.err || ""}`.slice(0, 200); return; }
      const f = await readFill(r.sig, ME, mint); upd(L.id, { tp_done: 1, tp_sol: f ? sol(f.solDelta) : null, tokens: (held - amount).toString(), reason: why }); balance = null; exportSoon(); return;
    }
    if (r.status !== "confirmed") { st.lastErr = `sell ${L.ticker}: ${r.status} ${r.err || ""}`.slice(0, 200); if (tries >= 6) upd(L.id, { status: "stuck", reason: "sell failed 6 times (press Sell to try again): " + String(r.err || r.status).slice(0, 120) }); return; }
    const fill = await readFill(r.sig, ME, mint); const got = fill ? sol(fill.solDelta) : null;
    upd(L.id, { status: "sold", sell_sig: r.sig, sell_sol: got, sold_ts: now(), pnl_sol: got != null && L.cost_sol != null ? +(got + (L.tp_sol || 0) + L.cost_sol).toFixed(6) : null });
    st.sells++; balance = null; exportSoon();
  } catch (e) { st.errors++; st.lastErr = "sell: " + e.message; }
}

        `Bundle: <b>${pct.toFixed(1)}%</b> of supply · <b>${bundleSol.toFixed(1)} SOL</b> · ${wallets.size} wallets (dev ${devBuy.toFixed(1)} SOL)\n` +
        `${socialsLine(soc)}\nMC now: $${fmt(mc)} · SOL $${solUsdCached().toFixed(0)}\n` +
        links(t.mint) + (t.factory ? "\n⚠️ factory: same creator launched another token in the last 10 min" : ""));
    }).catch(() => {});
  }
  setTimeout(() => expire(t), (t.tier >= 1 ? CFG.TRACK_MIN : CFG.TRACK_MIN_T0) * 60_000);
  emit("finalized", t);
}

export const holds = new Set(); // v0.18.119: mints other modules (Project S) want kept in memory past the usual 5 / 30 min
function expire(t) {
  if (t.hold || holds.has(t.mint)) return setTimeout(() => expire(t), 60_000); // open paper position keeps it alive
  if (t.volUntil && Math.floor(Date.now() / 1000) < t.volUntil) return setTimeout(() => expire(t), 60_000); // v0.19: a volume launch stays until its trades have been kept TRACK_MIN
  emit("expire", t); live.delete(t.mint);
}

export const links = (m) => `<a href="https://fomo.family/tokens/solana/${m}">FOMO</a> · <a href="https://axiom.trade/t/${m}">Axiom</a> · <a href="https://dexscreener.com/solana/${m}">DexScreener</a> · <a href="https://solscan.io/token/${m}">Solscan</a>\n<code>${m}</code>`;

export const fmt = (n) => (n >= 1e6 ? (n / 1e6).toFixed(2) + "M" : n >= 1e3 ? (n / 1e3).toFixed(1) + "K" : n.toFixed(0));
