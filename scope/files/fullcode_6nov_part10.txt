export function volumeQuality(w, seen = new Set()) {

  if (!n || !(vol > 0)) return null;

  // composite 0–100 (for the alert only; analysis uses the raw columns)
  let score = 0;
  score += Math.min(per10k / 20, 1) * 25;            // many wallets per $ → retail
  score += (1 - Math.min(top3, 1)) * 20;             // volume not concentrated
  score += Math.max(0, 1 - roundtrip / Math.max(wallets * 0.15, 1)) * 15; // few round-trippers
  score += Math.min(retail, 1) * 15;                 // via routers

function afterTrade(t, s, tr, isCluster) {

  if (!s.swDone && t.finalized && !t.graduated && now - t.ts > CFG.ENTRY_WINDOW_MIN * 60 && now - t.ts <= CFG.SW_WINDOW_MIN * 60 && s.wavePeakIn >= CFG.INFLOW_SOL) {

      }

  }

      if (t.lastMc >= CFG.CONF_MC_MULT * s.rawEntryMc && ratio >= CFG.CONF_BS_RATIO && buyersNow - s.buyersAtRaw >= CFG.CONF_BUYERS_GROW) {

// v0.18.120 — names the synchronous jobs that stall the event loop. timed(name, fn) runs fn and remembers it when it took over 300 ms;
// the stall monitor prints the last few, so a "[stall] blocked 9 s" line says what ran (CSV export, restart flags, page snapshot…) instead of leaving us guessing.
const recent = [];
export function timed(name, fn) { const t0 = Date.now(); try { return fn(); } finally { const dt = Date.now() - t0; if (dt > 300) { recent.push(`${name} ${dt} ms @${new Date(t0).toISOString().slice(11, 19)}Z`); if (recent.length > 6) recent.shift(); console.error(`[slow] ${name} took ${dt} ms`); } } }
export const slowJobs = () => (recent.length ? recent.join(" · ") : "none over 300 ms");
